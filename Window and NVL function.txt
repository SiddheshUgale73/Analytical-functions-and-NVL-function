--------------Windows/ Analytical function------------

rank()
dense_rank()
/*
-analytical functions are used to assign ranking numbers to each row or to each group of roe wise automatically.
-Rank(),dense_rank() are assigning same rank  number to same value but rank() will skip the next sequence rank number in the order 
   where as dense_rank() will not skip the next sequence rank number in the order.
-here ,
         partition by clause is optional.
         order by clause is mandatory.
         
   */

--ex: without using partition by clause 

select emp_name, salary, rank()
over 
(order by salary desc) ranks
from employees;

select emp_name, salary, dense_rank()
over 
(order by salary desc) ranks
from employees;


--ex : with using partition by clause 

select emp_name,salary, dept_id ,rank()
over 
(partition by dept_id order by salary desc )as ranks 
from employees;

select emp_name ,salary, dept_id, dense_rank()
over 
(partition by dept_id  order by salary desc) as ranks
from employees;

---query to find 4th highest salary from employee table 

select * from 
(select emp_name ,salary ,dept_id , dense_rank()
over 
(partition by dept_id order by salary desc)
as ranks from employees) 
where ranks=4;

---write a query to display 3rd senior .

select * from (select emp_name ,hire_date, dense_rank()
over (order by hire_date asc)
as senior from employees)
where  senior=3;

--------------------NVL function----------------

NVL() : The NVL() is used to replace with NULL value .It can take 2 Arguments .
NVL2 () : The NVL() is used to replace NULL or NOT NULL . It can take 3 arguments .

select nvl(NULL,'A')as Result from dual;

select nvl2(100,200,300)as result from dual;

select nvl2(100,null,300)as result from dual;

select nvl2(null,200,null)as result from dual;



----------------------VIEW in Sql---------
--view is the database object.
--view is a virtal table.it means it does not contains physical data.it does not occupy memory.
--it is created on tables .
--it holds select query, it does not hold query result.

-----Types of view 
1.simple view 
-- The view which is created based on one table is called simple view.
--it performs simple operations like insert , update, delete 

2.complex view 
--The view which is created based on multiple tables is called complex view.
---it performs complex operations like joins, group by ,having , aggregate functions etc..


create view empview as 
select emp_name,hire_date, dept_id from employees where salary>=60000;   ----simple view 

select * from empview;
